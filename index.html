/* ========================================= FRAGMENTED COLLAGE ENGINE Each photograph is a single logical group. Its fragments travel together to a new position, then reconstruct the photograph. No duplicate photographic layers. No unrelated floating pixel patches. ========================================= */ const canvas = document.getElementById("collage-canvas"); const ctx = canvas.getContext("2d",{alpha:false}); const collageWrap = document.querySelector(".collage-wrap"); let canvasWidth=1; let canvasHeight=1; let collageRunning=false; let collageReady=false; let lastFrame=0; let lastChange=0; const FRAME_INTERVAL=1000/24; const TRANSITION_DURATION=1750; const HOLD_DURATION=1600; /* Six photographs, six unique positions. Every new composition reassigns the photographs to different destinations. */ const compositions=[ [ {x:-.10,y:-.04,w:.62,h:.53}, {x:.48,y:-.06,w:.57,h:.46}, {x:.07,y:.46,w:.48,h:.39}, {x:.56,y:.40,w:.48,h:.40}, {x:-.05,y:.82,w:.42,h:.28}, {x:.36,y:.78,w:.59,h:.29} ], [ {x:.43,y:.05,w:.57,h:.49}, {x:-.07,y:.02,w:.53,h:.43}, {x:.51,y:.56,w:.51,h:.40}, {x:.02,y:.43,w:.53,h:.42}, {x:.33,y:.82,w:.47,h:.25}, {x:-.10,y:.78,w:.40,h:.31} ], [ {x:.06,y:.44,w:.57,h:.46}, {x:.44,y:-.07,w:.59,h:.48}, {x:-.09,y:.01,w:.52,h:.43}, {x:.57,y:.43,w:.46,h:.42}, {x:.28,y:.82,w:.49,h:.26}, {x:-.08,y:.76,w:.36,h:.29} ] ]; const collageGroups=[]; function resizeCanvas(){ const rect=collageWrap.getBoundingClientRect(); canvasWidth=Math.max(1,Math.round(rect.width)); canvasHeight=Math.max(1,Math.round(rect.height)); const ratio=Math.min( window.devicePixelRatio||1, 1.25 ); canvas.width=Math.round(canvasWidth*ratio); canvas.height=Math.round(canvasHeight*ratio); ctx.setTransform(ratio,0,0,ratio,0,0); collageReady=true; } function clamp(value,min,max){ return Math.max(min,Math.min(max,value)); } function lerp(a,b,t){ return a+(b-a)*t; } function easeInOut(t){ return t<.5 ? 4*t*t*t : 1-Math.pow(-2*t+2,3)/2; } function interpolateRect(a,b,t){ return { x:lerp(a.x,b.x,t), y:lerp(a.y,b.y,t), w:lerp(a.w,b.w,t), h:lerp(a.h,b.h,t) }; } /* Build a variable-size tile grid. Each photograph is partitioned into non-overlapping tiles. There are fewer large tiles and more small tiles. Every tile retains the correct portion of its original image. */ function createFragments(){ const fragments=[]; const rows=9; const cols=12; /* Uneven row and column boundaries. All tiles still fit together exactly. */ const widths=[]; const heights=[]; let totalW=0; let totalH=0; for(let c=0;cratio){ sh=ih; sw=ih*ratio; }else{ sw=iw; sh=iw/ratio; } return { x:(iw-sw)/2, y:(ih-sh)/2, w:sw, h:sh }; } const pixelCanvas=document.createElement("canvas"); const pixelCtx=pixelCanvas.getContext("2d"); /* Draw a single fragment from its exact corresponding region of the photograph. */ function drawFragment( image, crop, fragment, x,y,w,h, pixelated ){ const sx=crop.x+fragment.x*crop.w; const sy=crop.y+fragment.y*crop.h; const sw=fragment.w*crop.w; const sh=fragment.h*crop.h; if(pixelated){ const pw=Math.max( 1,Math.round(w/fragment.pixelSize) ); const ph=Math.max( 1,Math.round(h/fragment.pixelSize) ); pixelCanvas.width=pw; pixelCanvas.height=ph; pixelCtx.clearRect(0,0,pw,ph); pixelCtx.drawImage( image,sx,sy,sw,sh, 0,0,pw,ph ); ctx.save(); ctx.imageSmoothingEnabled=false; ctx.drawImage( pixelCanvas,x,y,w,h ); ctx.restore(); }else{ ctx.drawImage( image,sx,sy,sw,sh, x,y,w,h ); } } /* A fragment starts at its previous photograph position, scatters briefly, then joins the new photograph position. Once reconstruction finishes, it stays exactly in place. */ function drawGroup(group,time){ const image=group.image; if(!image.complete || !image.naturalWidth){ return; } const progress=group.moving ? clamp( (time-group.start)/group.duration, 0,1 ) : 1; const oldRect=group.from; const newRect=group.to; /* Crop using the destination image ratio. The same image portion is used for all fragments throughout movement. */ const crop=sourceCrop(image,newRect); ctx.save(); if(group.tone===1){ ctx.filter="grayscale(1) contrast(1.18)"; }else if(group.tone===2){ ctx.filter="saturate(1.25) contrast(1.12)"; } for(const fragment of group.fragments){ const delay=fragment.delay; const localProgress=group.moving ? clamp( (progress-delay)/(1-delay), 0,1 ) : 1; const t=easeInOut(localProgress); /* Position of this tile in its old and new photograph rectangles. */ const oldX= (oldRect.x+fragment.x*oldRect.w)* canvasWidth; const oldY= (oldRect.y+fragment.y*oldRect.h)* canvasHeight; const newX= (newRect.x+fragment.x*newRect.w)* canvasWidth; const newY= (newRect.y+fragment.y*newRect.h)* canvasHeight; const oldW= fragment.w*oldRect.w*canvasWidth; const oldH= fragment.h*oldRect.h*canvasHeight; const newW= fragment.w*newRect.w*canvasWidth; const newH= fragment.h*newRect.h*canvasHeight; /* Scatter peaks midway and disappears completely at the destination. */ const scatter= Math.sin(localProgress*Math.PI); const x= lerp(oldX,newX,t)+ fragment.scatterX* canvasWidth*scatter; const y= lerp(oldY,newY,t)+ fragment.scatterY* canvasHeight*scatter; const w=lerp(oldW,newW,t); const h=lerp(oldH,newH,t); /* Pixelation only during fragmentation. No permanent flying pixel rectangles. */ const pixelated= group.moving && localProgress>.05 && localProgress<.86 && fragment.pixelated; drawFragment( image, crop, fragment, x,y,w,h, pixelated ); } ctx.restore(); if(group.moving && progress>=1){ group.moving=false; group.from={...group.to}; group.current={...group.to}; } } /* The next composition uses one unique destination for each photograph. No image is drawn again at its old position after movement completes. */ let compositionIndex=0; function changeComposition(time){ compositionIndex= (compositionIndex+1)% compositions.length; const targets=compositions[compositionIndex]; collageGroups.forEach((group,index)=>{ group.from={...group.current}; group.to={...targets[index]}; group.start=time; group.duration=TRANSITION_DURATION; group.moving=true; /* New fragment pattern each transition. Only six groups are generated. */ group.fragments=createFragments(); }); } /* A very subtle grid across the collage. */ function drawGrid(){ ctx.save(); ctx.strokeStyle="rgba(10,20,35,.06)"; ctx.lineWidth=.5; const step=24; for(let x=0;x{ drawGroup(group,time); }); drawGrid(); if( !reducedMotion.matches && time-lastChange> TRANSITION_DURATION+HOLD_DURATION ){ changeComposition(time); lastChange=time; } } function startCollage(){ if(collageRunning) return; resizeCanvas(); collageRunning=true; requestAnimationFrame(renderCollage); } window.addEventListener("resize",()=>{ resizeCanvas(); renderCarousel(); });
